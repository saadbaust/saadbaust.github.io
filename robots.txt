# Allow everything except the gallery, which is intentionally private.
#
# SEO note: data.json and publications.bib are NOT blocked here, and that's
# intentional, not an oversight. Google's crawler renders the page's
# JavaScript before indexing content, which means it has to be able to fetch
# both files to see your real name, bio, experience, and publications (since
# all of that loads at runtime, not from static HTML). If you ever add a
# Disallow line for either file, Googlebot's renderer will fail to fetch it
# and large parts of the home page will effectively go un-indexed.
User-agent: *
Allow: /
Disallow: /gallery.html

# Belt-and-braces: also tell Google's image crawler to skip gallery photos.
# (The home-page hero charts under /images/ stay crawlable on purpose.)
User-agent: Googlebot-Image
Disallow: /images/events/
Disallow: /images/memories/

Sitemap: https://saadbaust.github.io/sitemap.xml
